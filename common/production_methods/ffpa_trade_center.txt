# 2050 Firefall Personal Adapter
# Vanilla 1.13 grants 10 Trade Capacity per fully employed Trade Center level.
# Injecting another 90 makes every level provide 100, regardless of the
# selected Trade Quantity production method.

INJECT:pm_trade_center = {
	state_modifiers = {
		workforce_scaled = {
			state_trade_capacity_add = 90
		}
	}
}

# External Trade II replaces pm_trade_center, so it needs the same injection.
INJECT:pm_trade_center_principle_external_trade_2 = {
	state_modifiers = {
		workforce_scaled = {
			state_trade_capacity_add = 90
		}
	}
}
